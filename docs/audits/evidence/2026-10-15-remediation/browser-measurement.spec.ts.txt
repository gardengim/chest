test('record bounded resources with populated thumbnail artwork',async({page})=>{
 await fixture(page);
 const session=await page.context().newCDPSession(page);
 const records=[];
 const record=async(label:string)=>{
  await session.send('HeapProfiler.collectGarbage');
  records.push({label,cards:await page.locator('.launcher-card').count(),nodes:await page.locator('*').count(),heapBytes:(await session.send('Runtime.getHeapUsage')).usedSize,iconCount:await page.locator('.instance-icon img[src^="data:"]').count()});
 };
 await expect(page.locator('.world-list .launcher-card')).toHaveCount(24);
 await page.locator('[data-home-section=servers]').scrollIntoViewIfNeeded();
 await expect(page.locator('.server-list .launcher-card')).toHaveCount(24);
 await expect(page.locator('.instance-icon img[src^="data:"]')).toHaveCount(100);
 await record('home-1000-worlds-1000-servers-100-thumbnails');
 await page.getByRole('button',{name:'Audit instance: Ready to play',exact:true}).click();
 await page.getByRole('tab',{name:'Contents',exact:true}).click();
 await expect(page.locator('.content-entry')).toHaveCount(24);
 await record('instance-1000-mods-100-thumbnails');
 await writeFile('/tmp/chest-browser-resources.json',JSON.stringify({boundary:'Chromium JS heap after requested GC; not total WebView memory, native CPU or a leak test',records},null,2));
});
